```rust
use crate::core::model::PrintSize;

pub const MM_PER_INCH: f32 = 25.4;

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum Preset { Five, Six, Seven }

pub fn preset(p: Preset) -> PrintSize {
    let (w, h) = match p {
        Preset::Five => (3.5, 5.0),
        Preset::Six => (4.0, 6.0),
        Preset::Seven => (5.0, 7.0),
    };
    PrintSize { width_mm: w * MM_PER_INCH, height_mm: h * MM_PER_INCH, dpi: 300 }
}

impl PrintSize {
    pub fn pixels(&self) -> (u32, u32) {
        let w = (self.width_mm / MM_PER_INCH * self.dpi as f32).round() as u32;
        let h = (self.height_mm / MM_PER_INCH * self.dpi as f32).round() as u32;
        (w.max(1), h.max(1))
    }
}
```

```rust
/// 返回源图裁切窗口 (x, y, w, h)。focus 0..1 控制裁切偏向。
pub fn fill_rect(
    src_w: u32,
    src_h: u32,
    target_w: u32,
    target_h: u32,
    focus: (f32, f32),
) -> (u32, u32, u32, u32) {
    let targ_ar = target_w as f32 / target_h as f32;
    let src_ar = src_w as f32 / src_h as f32;
    if src_ar > targ_ar {
        let new_w = ((src_h as f32 * targ_ar).round() as u32).clamp(1, src_w);
        let max_x = src_w - new_w;
        let x = (max_x as f32 * focus.0.clamp(0.0, 1.0)).round() as u32;
        (x, 0, new_w, src_h)
    } else {
        let new_h = ((src_w as f32 / targ_ar).round() as u32).clamp(1, src_h);
        let max_y = src_h - new_h;
        let y = (max_y as f32 * focus.1.clamp(0.0, 1.0)).round() as u32;
        (0, y, src_w, new_h)
    }
}

/// 等比缩放进 target，返回目标画布上的 (x, y, w, h)。
pub fn fit_rect(src_w: u32, src_h: u32, target_w: u32, target_h: u32) -> (u32, u32, u32, u32) {
    let scale = (target_w as f32 / src_w as f32).min(target_h as f32 / src_h as f32);
    let w = ((src_w as f32 * scale).round() as u32).clamp(1, target_w);
    let h = ((src_h as f32 * scale).round() as u32).clamp(1, target_h);
    ((target_w - w) / 2, (target_h - h) / 2, w, h)
}
```

```rust
use exif::{In, Reader, Tag, Value};
use std::io::Cursor;

#[derive(Debug, Clone)]
pub struct ExifInfo {
    pub datetime: Option<String>,
    pub gps: Option<(f64, f64)>,
    pub orientation: u16,
}

impl Default for ExifInfo {
    fn default() -> Self {
        Self { datetime: None, gps: None, orientation: 1 }
    }
}

pub fn gps_to_decimal(
    lat: (f64, f64, f64),
    lng: (f64, f64, f64),
    lat_ref: char,
    lng_ref: char,
) -> (f64, f64) {
    let to_deg = |d: (f64, f64, f64)| d.0 + d.1 / 60.0 + d.2 / 3600.0;
    let mut la = to_deg(lat);
    let mut ln = to_deg(lng);
    if lat_ref == 'S' { la = -la; }
    if lng_ref == 'W' { ln = -ln; }
    (la, ln)
}

fn rational_triplet(exif: &exif::Exif, tag: Tag) -> Option<(f64, f64, f64)> {
    match exif.get_field(tag, In::PRIMARY)?.value {
        Value::Rational(ref v) if v.len() >= 3 => {
            Some((v[0].to_f64(), v[1].to_f64(), v[2].to_f64()))
        }
        _ => None,
    }
}

fn ascii_char(field: &exif::Field) -> Option<char> {
    match &field.value {
        Value::Ascii(v) => v.first().and_then(|b| b.first()).map(|b| *b as char),
        _ => None,
    }
}

pub fn parse(bytes: &[u8]) -> ExifInfo {
    let mut info = ExifInfo::default();
    let exif = match Reader::new().read_from_container(&mut Cursor::new(bytes)) {
        Ok(e) => e,
        Err(_) => return info,
    };

    if let Some(f) = exif.get_field(Tag::DateTimeOriginal, In::PRIMARY) {
        info.datetime = Some(f.display_value().to_string().trim_matches('"').to_string());
    }

    if let (Some(lat), Some(lng)) = (
        rational_triplet(&exif, Tag::GPSLatitude),
        rational_triplet(&exif, Tag::GPSLongitude),
    ) {
        let lat_ref = exif
            .get_field(Tag::GPSLatitudeRef, In::PRIMARY)
            .and_then(ascii_char)
            .unwrap_or('N');
        let lng_ref = exif
            .get_field(Tag::GPSLongitudeRef, In::PRIMARY)
            .and_then(ascii_char)
            .unwrap_or('E');
        info.gps = Some(gps_to_decimal(lat, lng, lat_ref, lng_ref));
    }

    if let Some(f) = exif.get_field(Tag::Orientation, In::PRIMARY) {
        if let Value::Short(ref v) = f.value {
            if let Some(o) = v.first() {
                info.orientation = *o;
            }
        }
    }
    info
}
```

```rust
use serde::Deserialize;

#[derive(Debug, Clone, Deserialize)]
pub struct PlaceEntry {
    pub name: String,
    pub lat: f64,
    pub lng: f64,
}

pub struct PlaceDb {
    entries: Vec<PlaceEntry>,
}

fn haversine_km(lat1: f64, lng1: f64, lat2: f64, lng2: f64) -> f64 {
    let r = 6371.0;
    let dlat = (lat2 - lat1).to_radians();
    let dlng = (lng2 - lng1).to_radians();
    let a = (dlat / 2.0).sin().powi(2)
        + lat1.to_radians().cos() * lat2.to_radians().cos() * (dlng / 2.0).sin().powi(2);
    2.0 * r * a.sqrt().asin()
}

impl PlaceDb {
    pub fn from_entries(entries: Vec<PlaceEntry>) -> Self {
        Self { entries }
    }

    pub fn from_json(json: &str) -> Result<Self, serde_json::Error> {
        Ok(Self { entries: serde_json::from_str(json)? })
    }

    pub fn embedded() -> Self {
        const JSON: &str = include_str!("../../assets/geo/places.json");
        Self::from_json(JSON).expect("embedded place db is valid json")
    }

    pub fn nearest(&self, lat: f64, lng: f64) -> Option<&PlaceEntry> {
        self.entries
            .iter()
            .map(|e| (e, haversine_km(lat, lng, e.lat, e.lng)))
            .min_by(|a, b| a.1.partial_cmp(&b.1).unwrap())
            .map(|(e, _)| e)
    }
}
```
